 * ระบบแจ้งซ่อมออนไลน์ (Smart Maintenance System)
 * Backend: Google Apps Script — ทำงานเป็น JSON API เท่านั้น

// ============ ⚙️ CONFIG (แก้ไขค่าเหล่านี้ก่อนใช้งานจริง) ============
const SHEET_ID                    = 'YOUR_GOOGLE_SHEET_ID';           // ID ของ Google Sheet ที่ใช้เป็นฐานข้อมูล
const DRIVE_FOLDER_ID             = 'YOUR_DRIVE_FOLDER_ID';           // ID โฟลเดอร์ Google Drive สำหรับเก็บรูปภาพ
const LINE_CHANNEL_ACCESS_TOKEN   = 'YOUR_LINE_CHANNEL_ACCESS_TOKEN';  // Channel Access Token ของ LINE OA
const LINE_TARGET_ID              = 'YOUR_LINE_GROUP_OR_USER_ID';     // ปลายทางที่จะ push แจ้งเตือน (Group ID / User ID)

// อนุญาตเฉพาะโดเมนที่ระบุให้เรียก API ได้ (ใส่โดเมน GitHub Pages ของคุณ) — ใช้เพื่อกัน referer อื่นแบบหยาบๆ
// หมายเหตุ: Apps Script เว็บแอปเปิดให้เรียกแบบ public เสมอถ้าตั้ง Access เป็น "Anyone"
const ALLOWED_ORIGIN = '*';

const SHEET_NAMES = {
  REQUESTS:     'Requests',
  USERS:        'Users',
  TECHNICIANS:  'Technicians',
  DEPARTMENTS:  'Departments',
  EQUIPMENT:    'Equipment'
};

const REQUEST_HEADERS = [
  'TicketID', 'วันที่แจ้ง', 'LINE UserId', 'ชื่อผู้แจ้ง', 'ฝ่าย/สาขา',
  'อุปกรณ์', 'ความเร่งด่วน', 'รายละเอียดอาการ', 'รูปภาพ URL',
  'สถานะ', 'ช่างผู้รับผิดชอบ', 'วันที่อัปเดตล่าสุด'
];

// ============ 🌐 WEB APP ENTRY POINTS (JSON API) ============
// GET: ใช้สำหรับ action ที่ไม่มี payload หรือ payload เล็ก เช่น ?action=getDashboardData
// POST: ใช้สำหรับ action ที่มี payload เช่น submitMaintenanceRequest (ส่งรูปภาพ Base64)
function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  let action = '';
  let payload = {};

  try {
    if (e.postData && e.postData.contents) {
      // ฝั่งหน้าเว็บส่งมาแบบ Content-Type: text/plain เพื่อเลี่ยง CORS preflight
      // แล้วค่อย JSON.parse เนื้อหาที่นี่
      const body = JSON.parse(e.postData.contents);
      action = body.action;
      payload = body.payload || {};
    } else if (e.parameter) {
      action = e.parameter.action;
      payload = e.parameter.payload ? JSON.parse(e.parameter.payload) : {};
    }

    let result;
    switch (action) {
      case 'submitMaintenanceRequest':
        result = submitMaintenanceRequest(payload);
        break;
      case 'getDashboardData':
        result = getDashboardData();
        break;
      case 'getMasterData':
        result = getMasterData();
        break;
      case 'updateRequestStatus':
        result = updateRequestStatus(payload.ticketId, payload.status, payload.technician);
        break;
      case 'addMasterItem':
        result = addMasterItem(payload.sheetName, payload.value);
        break;
      case 'deleteMasterItem':
        result = deleteMasterItem(payload.sheetName, payload.rowIndex);
        break;
      case 'generateReportPdf':
        result = generateReportPdf();
        break;
      case 'ping':
        result = { success: true, message: 'API พร้อมใช้งาน' };
        break;
      default:
        result = { success: false, error: 'ไม่รู้จัก action: ' + action };
    }

    return jsonOutput(result);
  } catch (err) {
    return jsonOutput({ success: false, error: err.message });
  }
}

function jsonOutput(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// ============ 📄 SHEET HELPERS ============
function getSheet(name) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    if (name === SHEET_NAMES.REQUESTS) {
      sheet.appendRow(REQUEST_HEADERS);
      sheet.setFrozenRows(1);
    }
  }
  return sheet;
}

function generateTicketId() {
  const now = new Date();
  const ts = Utilities.formatDate(now, 'GMT+7', 'yyyyMMddHHmmss');
  return 'REQ-' + ts;
}

// ============ 📝 1) รับแจ้งซ่อม ============
function submitMaintenanceRequest(formObj) {
  try {
    const sheet = getSheet(SHEET_NAMES.REQUESTS);
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(REQUEST_HEADERS);
      sheet.setFrozenRows(1);
    }

    const ticketId = generateTicketId();
    let imageUrl = '';

    if (formObj.imageBase64 && formObj.imageBase64.length > 0) {
      imageUrl = saveImageToDrive(formObj.imageBase64, ticketId + '_' + (formObj.fileName || 'photo.jpg'));
    }

    const now = new Date();
    sheet.appendRow([
      ticketId,
      now,
      formObj.userId || '',
      formObj.fullName || '',
      formObj.department || '',
      formObj.equipment || '',
      formObj.urgency || 'ปกติ',
      formObj.detail || '',
      imageUrl,
      'รอดำเนินการ',
      '',
      now
    ]);

    // ส่งแจ้งเตือนเข้า LINE OA (ถ้าส่งไม่สำเร็จจะไม่ทำให้ทั้งฟังก์ชัน error)
    try {
      sendLineNotification(ticketId, formObj.fullName, formObj.department, formObj.equipment, formObj.urgency, formObj.detail, imageUrl);
    } catch (lineErr) {
      Logger.log('LINE notify error: ' + lineErr.message);
    }

    return { success: true, ticketId: ticketId };
  } catch (err) {
    return { success: false, error: err.message };
  }
}

// แปลง Base64 -> ไฟล์รูปภาพ แล้วบันทึกลง Google Drive
function saveImageToDrive(base64Data, fileName) {
  const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
  const parts = base64Data.split(',');
  const mimeMatch = parts.length > 1 ? parts[0].match(/data:(.*);base64/) : null;
  const mimeType = mimeMatch ? mimeMatch[1] : 'image/jpeg';
  const dataPart = parts.length > 1 ? parts[1] : parts[0];
  const bytes = Utilities.base64Decode(dataPart);
  const blob = Utilities.newBlob(bytes, mimeType, fileName);
  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return 'https://drive.google.com/uc?id=' + file.getId();
}

// ============ 📲 2) ส่งแจ้งเตือนผ่าน LINE Flex Message ============
function sendLineNotification(ticketId, fullName, department, equipment, urgency, detail, imageUrl) {
  let urgencyColor = '#28a745'; // ปกติ = เขียว
  if (urgency === 'ปานกลาง') urgencyColor = '#fd7e14'; // ส้ม
  if (urgency === 'สูง')     urgencyColor = '#dc3545'; // แดง

  const bodyContents = [
    { type: 'text', text: 'เลขที่: ' + ticketId, weight: 'bold', size: 'md' },
    { type: 'separator', margin: 'md' },
    { type: 'box', layout: 'baseline', margin: 'md', contents: [
      { type: 'text', text: 'ผู้แจ้ง', color: '#888888', size: 'sm', flex: 2 },
      { type: 'text', text: fullName || '-', wrap: true, size: 'sm', flex: 5 }
    ]},
    { type: 'box', layout: 'baseline', margin: 'sm', contents: [
      { type: 'text', text: 'ฝ่าย/สาขา', color: '#888888', size: 'sm', flex: 2 },
      { type: 'text', text: department || '-', wrap: true, size: 'sm', flex: 5 }
    ]},
    { type: 'box', layout: 'baseline', margin: 'sm', contents: [
      { type: 'text', text: 'อุปกรณ์', color: '#888888', size: 'sm', flex: 2 },
      { type: 'text', text: equipment || '-', wrap: true, size: 'sm', flex: 5 }
    ]},
    { type: 'box', layout: 'baseline', margin: 'sm', contents: [
      { type: 'text', text: 'เร่งด่วน', color: '#888888', size: 'sm', flex: 2 },
      { type: 'text', text: urgency || '-', color: urgencyColor, weight: 'bold', size: 'sm', flex: 5 }
    ]},
    { type: 'text', text: detail || '-', wrap: true, size: 'sm', color: '#555555', margin: 'md' }
  ];

  const bubble = {
    type: 'bubble',
    header: {
      type: 'box', layout: 'vertical', backgroundColor: '#28a745', paddingAll: '16px',
      contents: [
        { type: 'text', text: '🛠 แจ้งซ่อมใหม่เข้าระบบ', weight: 'bold', color: '#ffffff', size: 'lg' }
      ]
    },
    body: { type: 'box', layout: 'vertical', contents: bodyContents, paddingAll: '16px' }
  };

  if (imageUrl) {
    bubble.footer = {
      type: 'box', layout: 'vertical', paddingAll: '12px',
      contents: [{
        type: 'button', style: 'primary', color: '#fd7e14',
        action: { type: 'uri', label: '📷 ดูรูปภาพ', uri: imageUrl }
      }]
    };
  }

  const payload = {
    to: LINE_TARGET_ID,
    messages: [{ type: 'flex', altText: '🛠 แจ้งซ่อมใหม่ ' + ticketId, contents: bubble }]
  };

  UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + LINE_CHANNEL_ACCESS_TOKEN },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });
}

// ============ 📊 3) ข้อมูลสำหรับ Dashboard ============
function getDashboardData() {
  const sheet = getSheet(SHEET_NAMES.REQUESTS);
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    return { requests: [], stats: { total: 0, pending: 0, done: 0 }, byStatus: {}, byDept: {} };
  }

  const rows = data.slice(1);
  const requests = rows.map(r => ({
    ticketId: r[0],
    date: r[1] instanceof Date ? r[1].toISOString() : r[1],
    userId: r[2], fullName: r[3], department: r[4], equipment: r[5],
    urgency: r[6], detail: r[7], imageUrl: r[8], status: r[9],
    technician: r[10],
    updatedAt: r[11] instanceof Date ? r[11].toISOString() : r[11]
  })).filter(r => r.ticketId);

  const stats = { total: requests.length, pending: 0, done: 0 };
  const byStatus = {};
  const byDept = {};

  requests.forEach(r => {
    if (r.status === 'เสร็จสิ้น') stats.done++; else stats.pending++;
    byStatus[r.status] = (byStatus[r.status] || 0) + 1;
    byDept[r.department] = (byDept[r.department] || 0) + 1;
  });

  return { requests: requests.reverse(), stats: stats, byStatus: byStatus, byDept: byDept };
}

// ============ 🛠️ ADMIN: จัดการสถานะงานซ่อม ============
function updateRequestStatus(ticketId, status, technician) {
  const sheet = getSheet(SHEET_NAMES.REQUESTS);
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === ticketId) {
      sheet.getRange(i + 1, 10).setValue(status);
      if (technician) sheet.getRange(i + 1, 11).setValue(technician);
      sheet.getRange(i + 1, 12).setValue(new Date());
      return { success: true };
    }
  }
  return { success: false, error: 'ไม่พบ Ticket นี้ในระบบ' };
}

// ============ 🛠️ ADMIN: ข้อมูลตั้งต้น (ผู้ใช้งาน/ช่าง/ฝ่าย/อุปกรณ์) ============
function getMasterData() {
  return {
    departments: flattenColumn(getSheet(SHEET_NAMES.DEPARTMENTS)),
    technicians: flattenColumn(getSheet(SHEET_NAMES.TECHNICIANS)),
    equipment:   flattenColumn(getSheet(SHEET_NAMES.EQUIPMENT)),
    users:       flattenColumn(getSheet(SHEET_NAMES.USERS))
  };
}

function flattenColumn(sheet) {
  const values = sheet.getDataRange().getValues();
  return values.map(r => r[0]).filter(v => v !== '' && v !== null);
}

function addMasterItem(sheetName, value) {
  const sheet = getSheet(sheetName);
  sheet.appendRow([value]);
  return { success: true };
}

function deleteMasterItem(sheetName, rowIndex) {
  const sheet = getSheet(sheetName);
  sheet.deleteRow(rowIndex); // rowIndex คือแถวจริงใน Sheet (เริ่มที่ 1)
  return { success: true };
}

// ============ 📑 ADMIN: สร้างรายงาน PDF ============
function generateReportPdf() {
  const dash = getDashboardData();
  const requests = dash.requests;

  const summarize = (keyFn) => {
    const map = {};
    requests.forEach(r => {
      const k = keyFn(r) || '(ไม่ระบุ)';
      map[k] = (map[k] || 0) + 1;
    });
    return map;
  };

  const byStatus     = summarize(r => r.status);
  const byDept       = summarize(r => r.department);
  const byEquipment  = summarize(r => r.equipment);
  const byTechnician = summarize(r => r.technician || 'ยังไม่ได้มอบหมาย');
  const byMonth      = summarize(r => {
    const d = new Date(r.date);
    return Utilities.formatDate(d, 'GMT+7', 'MM/yyyy');
  });

  const html = buildReportHtml(requests, byStatus, byDept, byEquipment, byMonth, byTechnician);
  const blob = Utilities.newBlob(html, 'text/html', 'report.html').getAs('application/pdf');
  blob.setName('รายงานแจ้งซ่อม_' + Utilities.formatDate(new Date(), 'GMT+7', 'yyyyMMdd_HHmmss') + '.pdf');

  return {
    success: true,
    base64: Utilities.base64Encode(blob.getBytes()),
    fileName: blob.getName()
  };
}

function buildReportHtml(requests, byStatus, byDept, byEquipment, byMonth, byTechnician) {
  const tableRows = (map) => Object.keys(map).map(k =>
    '<tr><td>' + k + '</td><td>' + map[k] + '</td></tr>').join('');

  const requestRows = requests.map(r =>
    '<tr><td>' + r.ticketId + '</td><td>' + r.fullName + '</td><td>' + r.department +
    '</td><td>' + r.equipment + '</td><td>' + r.urgency + '</td><td>' + r.status + '</td></tr>'
  ).join('');

  return `
  <html><head><meta charset="utf-8"><style>
    body{font-family:'TH Sarabun New','Sarabun',sans-serif;font-size:14px;color:#222}
    h1{color:#28a745;font-size:22px} h2{color:#fd7e14;font-size:17px;margin-top:24px}
    table{width:100%;border-collapse:collapse;margin-top:8px}
    th,td{border:1px solid #ccc;padding:6px 8px;text-align:left;font-size:12px}
    th{background:#28a745;color:#fff}
  </style></head><body>
    <h1>รายงานสรุประบบแจ้งซ่อมออนไลน์</h1>
    <p>วันที่ออกรายงาน: ${Utilities.formatDate(new Date(), 'GMT+7', 'dd/MM/yyyy HH:mm')}</p>

    <h2>สรุปตามสถานะ</h2>
    <table><tr><th>สถานะ</th><th>จำนวน</th></tr>${tableRows(byStatus)}</table>

    <h2>สรุปตามฝ่าย/สาขา</h2>
    <table><tr><th>ฝ่าย/สาขา</th><th>จำนวน</th></tr>${tableRows(byDept)}</table>

    <h2>สรุปตามอุปกรณ์</h2>
    <table><tr><th>อุปกรณ์</th><th>จำนวน</th></tr>${tableRows(byEquipment)}</table>

    <h2>สรุปตามเดือน</h2>
    <table><tr><th>เดือน/ปี</th><th>จำนวน</th></tr>${tableRows(byMonth)}</table>

    <h2>สรุปตามช่างซ่อม</h2>
    <table><tr><th>ช่างซ่อม</th><th>จำนวน</th></tr>${tableRows(byTechnician)}</table>

    <h2>รายการแจ้งซ่อมทั้งหมด</h2>
    <table>
      <tr><th>Ticket</th><th>ผู้แจ้ง</th><th>ฝ่าย/สาขา</th><th>อุปกรณ์</th><th>เร่งด่วน</th><th>สถานะ</th></tr>
      ${requestRows}
    </table>
  </body></html>`;
}
